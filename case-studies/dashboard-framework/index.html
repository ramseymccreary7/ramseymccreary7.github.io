Reusable Dashboard Framework

Every dashboard on this site sits on the same foundation: email one-time-code sign-in, an access list, server-side field allow-lists, and row-level filtering by role and office. I packaged it as templates and a build skill so each new dashboard starts out secure.

How it flows

What I built

  • Sign-in uses an emailed 6-digit code, so viewers need no Google account. Codes are throttled per email and capped per hour globally, wrong guesses are limited, and the page never reveals whether an email is on the list.
  • Sessions are opaque random tokens with an expiry, checked on every server call along with per-person rate limits.
  • Payloads are built from an explicit field allow-list, so internal columns can't leak even if the sheet grows, and a second check trips on anything that should never leave the server.
  • An audit log records sign-ins, views and denials. The kit includes templates, an access-tab spec, render tests and a build checklist.

Outcome

  • More than ten production dashboards share it, from internal finance tools to an external partner portal.

function requestAccessCode(email) {
  email = String(email || '').toLowerCase().trim();
  var cache = CacheService.getScriptCache();
  try {
    if (!EMAIL_RE_.test(email)) return { ok: true };
    if (cache.get('THROTTLE_' + email)) return { ok: true };

    var used = Number(cache.get('GLOBAL_CODES') || 0);
    if (used >= GLOBAL_CODES_PER_HOUR) {   // a drained mail quota also kills notifications
      audit_('code_global_cap', email, '', 'global hourly cap reached');
      return { ok: true };
    }

    var access = getAccessMap_();
    if (!access[email]) return { ok: true };   // silent: do not confirm membership

    var code = sixDigitCode_();
    cache.put('CODE_' + email, code, CODE_TTL_S);
    cache.put('THROTTLE_' + email, '1', THROTTLE_S);
    cache.put('GLOBAL_CODES', String(used + 1), 3600);
    // NOTE: do NOT reset ATT_ here. Resetting it turns a 6-attempt cap into
    // 6 attempts per throttle window, forever.

    MailApp.sendEmail({
      to: email,
      subject: CONFIG.title + ' — your access code',
      body: 'Your access code is ' + code + '\n\nIt expires in 10 minutes.\n\n'
          + 'If you did not request this, you can ignore this email.'
    });
    audit_('code_sent', email, '', '');
  } catch (err) {
    audit_('code_send_failed', email, '', String(err));
    try {
      MailApp.sendEmail(CONFIG.notifyEmail,
        CONFIG.title + ' — access code send FAILED',
        'Address: ' + email + '\nError: ' + String(err));
    } catch (e2) {}
  }
  return { ok: true };     // same answer on every path
}

function verifyAccessCode(email, code) {
  email = String(email || '').toLowerCase().trim();
  var cache = CacheService.getScriptCache();
  var att   = Number(cache.get('ATT_' + email) || 0);
  if (att >= MAX_ATTEMPTS) { audit_('code_attempts_exceeded', email, '', ''); return { ok: false }; }
  cache.put('ATT_' + email, String(att + 1), CODE_TTL_S);

  var want = cache.get('CODE_' + email);
  if (!want || String(code).trim() !== String(want)) return { ok: false };

  var access = getAccessMap_();
  if (!access[email]) return { ok: false };

  var token = Utilities.getUuid();
  var hours = Math.min(CONFIG.sessionHours, CONFIG.sessionHoursMax);
  PropertiesService.getScriptProperties()
    .setProperty(tokenKey_(token), email + '|' + (Date.now() + hours * 3600 * 1000));
  cache.remove('CODE_' + email);
  cache.remove('ATT_' + email);
  audit_('sign_in', email, '', 'otp');
  return { ok: true, token: token, email: email };
}
