Commission & bonus checks

Payroll Validation

Agent pay depends on deal profit, state base rates, split rules and trailing-three-month bonus grids. This tool recomputes every submitted payroll record and flags anything that doesn't tie.

Google Apps ScriptSalesforce (SOQL + field history)Claude API fallback

How it flows

What I built

  • Validates every submitted base and bonus record against the closed deal's documents and CRM values.
  • Base pay uses the state rate times profit, with splits for secondary agents. Bonuses come from a trailing-three-month units-by-volume grid, and units are rounded down so a tier must be fully earned.
  • An append-only validation log is keyed on a hash of the paid values, so a record that changes after validation shows as "Changed since validation" instead of a stale green check.
  • Feeds the Deal Payroll tab of the GM portal, including estimated bonuses and a request cutoff with a live countdown.

Outcome

  • Payroll is reviewed against the source documents before it posts.
  • GMs see the same numbers as Accounting, with the math one hover away.

/**
 * The grid's total rate for a T3M average. Units are rounded down; the highest units row ≤ that and the highest volume
 * bracket ≤ the average volume. → {rate, units, volIdx, label} or {rate: null, why}.
 */
function pvGridRate_(grids, type, state, avg) {
  var u = Math.floor((type === 'aa' ? avg.aaU : avg.daU) + 1e-9);
  if (type === 'aa') {
    var g = grids.aa[state];
    if (!g || !g.rows.length) return { rate: null, why: 'No AA bonus grid for ' + (state || 'unknown state') + ' (' + PV.RATES_TAB + ' E:L)' };
    var row = null;
    g.rows.forEach(function (r) { if (r.units <= u) row = r; });
    if (!row) return { rate: null, below: true, units: u, why: 'below the first units row (' + g.rows[0].units + ')' };
    var vi = 0;
    g.vols.forEach(function (t, i) { if (avg.aaV + 1e-9 >= t) vi = i; });
    while (vi > 0 && (row.rates[vi] === null || row.rates[vi] === undefined)) vi--;
    if (row.rates[vi] === null || row.rates[vi] === undefined) return { rate: null, why: 'blank rate in the ' + state + ' grid' };
    var vLabel = g.vols[vi] === 0 ? 'under ' + pvFmt0_(g.vols[1] || 0) : pvFmt0_(g.vols[vi]) + '+';
    return { rate: row.rates[vi], units: row.units, label: state + ' AA grid: ' + row.units + ' unit row × ' + vLabel + ' volume column' };
  }
  var tiers = grids.da[state];
  if (!tiers || !tiers.length) return { rate: null, why: 'No DA bonus tiers for ' + (state || 'unknown state') + ' (' + PV.RATES_TAB + ' N:Q)' };
  var tier = null;
  tiers.forEach(function (t) { if (t.units <= u) tier = t; });
  if (!tier) return { rate: null, below: true, units: u, why: 'below the first tier (' + tiers[0].units + ' units)' };
  return { rate: tier.rate, units: tier.units, label: state + ' DA tier ' + (tier.tier || tier.units + ' units') };
}
function pvFmt0_(n) { return '$' + Math.round(n).toLocaleString('en-US'); }
